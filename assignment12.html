  1. (Model Comparison) Use the PimaIndiansDiabetes dataset. Pick two models at this link to compare using 7-fold cross validation method. Evaluate the accuracy of the final model on the test data.
library(tidyverse)
library(caret)
library(mlbench)
library(rpart)
library(ranger)

data(PimaIndiansDiabetes, package = "mlbench")

df <- tibble(PimaIndiansDiabetes)
df$diabetes <- as.factor(df$diabetes)

# split
set.seed(123)
splitIndex <- createDataPartition(df$diabetes, p = 0.85, list = FALSE)

df_train <- df[splitIndex, ]
df_test <- df[-splitIndex, ]

# folds
trControl <- trainControl(method = "cv", number = 7)

# decision tree
set.seed(123)
tree_model <- train(diabetes ~ ., data = df_train, method = "rpart2",
trControl = trControl, tuneGrid = expand.grid(maxdepth = 2:10),
metric = "Accuracy")

# random forest
set.seed(123)
forest_model <- train(diabetes ~ ., data = df_train, method = "ranger",
trControl = trControl, metric = "Accuracy")

# compare
results <- resamples(list("Decision Tree" = tree_model,          "Random Forest" = forest_model))

bwplot(results)
